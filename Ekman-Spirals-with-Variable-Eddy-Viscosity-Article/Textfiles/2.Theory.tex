\section{Theory}
\subsection{Problem formulation}
The Ekman equation is derived from the steady-state Navier--Stokes equations on a rotating frame with shear flow. Assuming friction, incompressible flow, negligible vertical motion, gives the horizontal momentum equations as
\begin{subequations}
\begin{empheq}[left=\empheqlbrace]{align}
    -fv &= -\frac{1}{\rho}\frac{\mathrm{d} p}{\mathrm{d} x}
    + \frac{\mathrm{d}}{\mathrm{d} z}\left(\nu \frac{\mathrm{d} u}{\mathrm{d} z}\right), \\
    fu &= -\frac{1}{\rho}\frac{\mathrm{d} p}{\mathrm{d} y}
    + \frac{\mathrm{d}}{\mathrm{d} z}\left(\nu \frac{\mathrm{d} v}{\mathrm{d} z}\right),
    \label{eq:NavierStokes}
\end{empheq}
\end{subequations}
where $u$ and $v$ are the horizontal velocity components, $f$ is the Coriolis parameter, $\rho$ is the fluid density, and $\nu$ is the viscosity. Far away from the surface, shear flow and friction becomes negligible and the flow will thus approach geostrophic balance, $(u,v)\rightarrow (u_g,v_g)$, as were one can neglect the friction term in \autoref{eq:NavierStokes}. Subtracting the geostrophic balance from \autoref{eq:NavierStokes} thus gives
\begin{subequations}
\begin{empheq}[left=\empheqlbrace]{align}
    -f(v-v_g) &=
    \frac{\mathrm{d}}{\mathrm{d} z}\left(\nu \frac{\mathrm{d} u}{\mathrm{d} z}\right), \\
    f(u-u_g) &=
    \frac{\mathrm{d}}{\mathrm{d} z}\left(\nu \frac{\mathrm{d} v}{\mathrm{d} z}\right).
    \label{eq:subtractedNavierStokes}
\end{empheq}
\end{subequations}
This problem can be formulated with one equation if one introduces a complex velocity defined as 
\begin{align}
    U \equiv u + iv,
    \label{eq:ComplexVelocity}
\end{align}
and a complex geostrophic velocity as $U_g = u_g + iv_g$. Using \autoref{eq:subtractedNavierStokes} with the complex velocities $U$ and $U_g$ the coupled equations can be combined into the Ekman equation
\begin{empheq}{equation}
    \frac{\mathrm{d}}{\mathrm{d} z}
    \left(
    \nu \frac{\mathrm{d} U}{\mathrm{d} z}
    \right)
    =
    if(U-U_g).
    \label{eq:EkmanEquation}
\end{empheq}
With the boundary conditions
\begin{subequations}
\begin{align}
    U(0) &= 0,
    \label{eq:BoundaryCondition1}
    \\
    U(z\to\infty) &= U_g.
    \label{eq:BoundaryCondition2}
\end{align}
\label{eq:BoundaryConditions}
\end{subequations}

\textbf{ANGLE AND TRANSPORT?}

\subsection{The vertical momentum flux}
The Ekman equation can be reformulated in terms by introducing a complex vertical momentum flux as
\begin{equation}
    \tau=\nu\frac{\mathrm{d}U}{\mathrm{d}z}.
    \label{eq:UtoTau}
\end{equation}
Using this definition, \autoref{eq:EkmanEquation} can be rewritten as
\begin{empheq}[box=\fbox]{equation}
    \frac{\mathrm{d}^2\tau}{\mathrm{d}z^2}=i\underbrace{\lambda}_{\equiv \frac{f}{\nu}}
    \tau.
    \label{eq:MomentumFlux}
\end{empheq}
Here, an effective potential $\lambda$ has been introduced. This allows \autoref{eq:MomentumFlux} to be compared with the time-independent Schrödinger equation, where $\lambda$ becomes analogous to a potential. The new equation also has a clear physical interpretation, as it describes the vertical structure of the momentum flux in the classical Ekman problem. Using \autoref{eq:MomentumFlux}, the boundary conditions become
\begin{subequations}
\begin{align}
    \left.\frac{\mathrm{d}\tau}{\mathrm{d}z}\right|_{z=0} &= -ifU_g,
    \label{eq:MomentumFluxBoundaryConditions1}
    \\
    \tau(z\to\infty) &= 0.
    \label{eq:MomentumFluxBoundaryConditions2}
\end{align}
\label{eq:MomentumFluxBoundaryConditions}
\end{subequations}

\subsection{The classical solution}
The classical solution to the Ekman equation is obtained when the viscosity is assumed to be constant with altitude, $\nu=\text{constant}$, and can be found in \citet{Ekman1905}. In this case he classical Ekman spiral solution from \autoref{eq:EkmanEquation} can obtained as
\begin{equation}
    \tau(z)=\nu U_g \frac{1+i}{h_\text{Ek}} e^{- \frac{1+i}{h_\text{Ek}} z},
    \label{eq:EkmanConstant}
\end{equation}
where
\begin{equation}
    h_\text{Ek}\equiv\sqrt{\frac{2\nu}{f}}
    \label{eq:EkmanHeight}
\end{equation}
is the Ekman depth scale. One can note that by taking the argument of the complex momentum flux, $\arg\left[\tau(z=0)\right]$, one can measure the surface angle of \autoref{eq:EkmanConstant} as \ang{45}.

\subsection{The two layer model}
The next natural step from a constant viscosity profile is to assume two layers of constant viscosity in each layer. One can label the lower level as having the viscosity $\nu_1$ up until a height of $H$, where the next viscosity of $\nu_2$ takes over. From this assumption one would obtain a superimposed solution similar to \autoref{eq:EkmanConstant} as 
\begin{subequations}
\begin{empheq}[left=\empheqlbrace]{align}
    \tau_1(z)&=A_1e^{\frac{1+i}{h_\text{Ek1}}z}+B_1e^{-\frac{1+i}{h_\text{Ek1}}z}, \\
    \tau_2(z)&=B_2e^{-\frac{1+i}{h_\text{Ek2}}z}
\end{empheq}
\end{subequations}
where the Ekman depth scales are defined for each layer like in \autoref{eq:EkmanHeight}. To solve for the constants above one has to use the fact that the material flux has to be constant. This gives the jump condition as 
\begin{subequations}
\begin{align}
    U_1&=U_2 \quad \text{at } z=H, \\
    \tau_1&=\tau_2  \quad \text{at } z=H.
    \label{eq:JumpCondition}
\end{align}
\end{subequations}
Using this and the boundary conditions at \autoref{eq:BoundaryConditions} one obtains a expression between for factors as
\begin{subequations}
\begin{align}
    A_1&=-\nu_1U_g\frac{1+i}{h_\text{Ek1}}\cdot\frac{1-\gamma}{1-\gamma+e^{2\left(1+i\right)\varphi}\left(1+\gamma\right)}, \text{ where }\gamma=\sqrt{\tfrac{\nu_2}{\nu_1}} \\ 
    B_1&=\nu_1U_g\frac{1+i}{h_\text{Ek1}}\cdot\frac{\left(1+\gamma\right)e^{2\left(1+i\right)\varphi}}{1-\gamma+e^{2\left(1+i\right)\varphi}\left(1+\gamma\right)}, \\ 
    B_2&=\nu_2U_g\frac{1+i}{h_\text{Ek2}}\cdot\frac{2e^{\left(1+i\right)\varphi\left(1+\gamma^{-1}\right)}}{1-\gamma+e^{2\left(1+i\right)\varphi}\left(1+\gamma\right)},
\end{align}
\end{subequations}
where $\varphi$ is the non-dimensional layer thickness defined as 
\begin{equation}
    \varphi = \frac{H}{h_\text{Ek1}}.
\end{equation}
The constant $R$ is defined as 
\begin{equation}
    R = \frac{1-\gamma}{1+\gamma}.
\end{equation}

This gives a full expression for the vertical momentum flux in the two layer model and can thus be plotted as seen in \autoref{fig:2layer_structure}. There one can observe how vertical momentum flux changes with altitude for different viscosity ratios and non-dimensional heights, $\varphi$.

    \centering    
    \includegraphics[height=0.44\textwidth]{Figures/2layer_structure.png}
    \caption{The vertical structure of the momentum flux for a two layer viscosity. The momentum flux is dependent on the dimensionless parameter $\varphi=H/h_\text{Ek1}$and the ratio of the viscosities. The corresponding Ekman heights and viscosity potential is shown with the shown with the vertical momentum flux. The solution assumed a purely zonal geostrophic flow.}
    \label{fig:2layer_structure}
\end{figure}

Using the derived solution for the vertical momentum flux, one can derive the surface angle as 
\begin{align}
    \theta=\arg \left[\tau(z=0)\right] &= \arg\left[\left(A_1+B_1\right)\right] \\
    &=45\degr + 
    \arg\left[\frac{e^{(1+i)\varphi}-Re^{-(1+i)\varphi}}{e^{(1+i)\varphi}+Re^{-(1+i)\varphi}}\right]
    \label{eq:2layerAngle}
\end{align}

The result from the theoretical two layer model, \autoref{eq:2layerAngle}, can be seen in \autoref{fig:2layer_angle}. There one can observe solutions for different viscosity ratios, half of which have an higher upper-level viscosity and the other half a lower one. These are plotted together with a \ang{45} reference point to highlight the difference from the classical Ekman solution. One can observe a convergence towards the classical Ekman value when $\varphi\rightarrow\infty$ for all the viscosity ratios. The zero limit is, however, dependent on the viscosity ratio, where finite ratios approach \ang{45}. However, when $\tfrac{\nu_2}{\nu_1}\rightarrow0$ and $\varphi\rightarrow 0$ one can observe that $\theta\rightarrow\ang{90}$. Respectively, one can observe that when $\tfrac{\nu_2}{\nu_1}\rightarrow\infty$ and $\varphi\rightarrow0$, $\theta\rightarrow\ang{0}$. One can note that for $\nu_1>\nu_2$ (and $\nu_2>\nu_1$ respectively) the solution will always provide a result greater than \ang{45}, except between the $\varphi\in\left(\tfrac{\pi}{2},\pi\right)$ were a angle below \ang{45} can be observed.

\end{figure}

\subsection{The exponential decaying viscosity}
Another viscosity profile that can provide a fully analytical solution is an exponentially decaying viscosity with altitude. Here the viscosity is set to decay from an initial viscosity of $\nu_0$, with a decay rate of $k$ as $\nu(z)=\nu_0\exp\left(-kz\right)$. To solve \autoref{eq:EkmanEquation} with this viscosity profile one can introduce two new parameters 
\begin{equation}
    t=4\left(1+i\right)\,\varphi_\text{exp}\, e^{\frac{kz}{4}} 
\end{equation}
The nondimensional parameter $\varphi_\text{exp}$ was introduced above in this problem, just like in the 2-layer model, but here it is defined as 
\begin{equation}
    \varphi_\text{exp}=\frac{1}{k\,h_\text{Ek0}}=\frac{1}{k}\sqrt{\frac{f}{2\nu_0}}.
\end{equation}
Using these newly defined variables \autoref{eq:EkmanEquation} transforms into 
\begin{equation}
    t^2\frac{\mathrm{d}^2 \tau}{\mathrm{d} t^2}+t\frac{\mathrm{d} \tau}{\mathrm{d} t}-t^2\tau=0,
\end{equation}
which is a differential equation with a standard solution. The standard solution is a modified Bessel equation as 
\begin{equation}
    \tau=C_1I_0(t)+C_2K_0(t),
\end{equation}
where $C_1$ and $C_2$ are constants, while $I_0(t)$ and $K_0(t)$ are well-defined modified Bessel functions of the first and second kind of order zero, respectively. By applying the previously defined boundary conditions from \autoref{eq:MomentumFluxBoundaryConditions} one can find the two constants. Applying the geostrophic boundary condition from \autoref{eq:MomentumFluxBoundaryConditions2} one can see that $C_1=0$ since $I_0(t)$ does not go to zero when $t\to\infty$ (which occurs when $z\to\infty$). Applying the surface boundary condition from \autoref{eq:MomentumFluxBoundaryConditions1} one can find an expression for the second constant. Transforming the solution to the normal complex velocity as in \autoref{eq:ComplexVelocity} one can obtain the solution
\begin{equation}
    \tau(z)=\frac{\left(1+i\right)fU_g}
    {2k\varphi}
    \frac{K_0\left(4\left[1+i\right]\varphi_\text{exp} e^{kz/4}\right)}{K_1\left(4\left[1+i\right]\varphi_\text{exp}\right)}
    \label{eq:exponetial}
\end{equation}
This gives a full expression for the vertical momentum flux in the exponential model and as seen in \autoref{fig:2layer_structure}. There one can observe how vertical momentum flux changes with altitude for different non-dimensional heights, $\varphi$. 
\begin{figure}[H]
    \centering    
    \includegraphics[height=0.44\textwidth]{Figures/exponential_structure.png}
    \caption{The vertical structure of the momentum flux for exponentially decreasing viscosity. The momentum flux is dependent on the dimensionless parameter $\varphi_\text{exp}=1/kh_\text{Ek0}$. The corresponding Ekman heights and viscosity potential is shown with the shown with the vertical momentum flux. The solution assumed a purely zonal geostrophic flow.}
    \label{fig:exponential_structure}
\end{figure}

Using \autoref{eq:exponetial} one can derive the surface angle of this model as 
\begin{equation}
    \theta = \arg \left[\tau(z=0)\right] = \ang{45} + \arg\left[\frac{K_0\left(4\left[1+i\right]\,\varphi_\text{exp}\,\right)}{K_1\left(4\left[1+i\right]\,\varphi_\text{exp}\,\right)}\right],
    \label{eq:ExponetialAngle}
\end{equation} 
where $K_1$ is the modified Bessel function of the second kind of order one. 

\autoref{fig:exponential_angle} showcases the result from the exponentially decaying viscosity as in \autoref{eq:ExponetialAngle}. There one can observe a similar limit as in \autoref{fig:2layer_angle} when $\varphi\rightarrow 0$, as \autoref{eq:ExponetialAngle} also approaches \ang{90}. The same limit is also true for $\varphi\rightarrow\infty$, which gives $\theta\rightarrow\ang{45}$, although this exponential solution converges more slowly than the two-layer solution. However, the shape is much more uniform, with an exponential-like decay. However, all angels are here greater then the reference angle of \ang{45}.

\subsection{The simplified upper boundary condition model}
In the previous sections two completely analytical solutions for two different altitude-varying viscosity profiles were derived. The difficulty in solving \autoref{eq:MomentumFlux} comes from the altitude dependence of the viscosity profile $\nu(z)$. This introduces variable coefficients into the complex second-order ordinary differential equation, which in general prevents closed-form analytical solutions. Analytical solutions therefore exist only for specific viscosity profiles that allow the equation to be transformed into a solvable form.

\autoref{eq:MomentumFlux} is furthermore also difficult to solve in a numerical setting, as the upper boundary condition is a limit. This will put limits on the viscosity profiles one can study, as the viscosity cannot increase with altitude continuously. In the physical world, this makes sense, since we assume that high up in the atmosphere (or deep down in the ocean), friction and shear stress become negligible. This was what was labelled as the geostrophic limit. However, by limiting ourselves to only upper-level decreasing viscosity profiles, the differential equation becomes more difficult to solve. Furthermore, if one wants to solve for a dimensionless height parameter, too complex viscosity profiles become hard to deal with.

A solution to this problem is to introduce the simplified upper boundary condition model. For the 2-layer model, this is reduced to a 1.5 layer model, since we assume that the upper-level viscosity is negligible $\nu_2\approx0$. This also makes physical sense, since we know that both the atmosphere and the ocean consist of a boundary layer where turbulent mixing takes place. Reducing the problem to a simplified upper boundary condition model will thus make it possible to test viscosity profiles that do not converge at $z\rightarrow\infty$, since the new domain is between $z\in[0,H]$. This modifies the boundary conditions in \autoref{eq:MomentumFluxBoundaryConditions} as 
\begin{subequations}
\label{eq:NewBoundaryConditions}
\begin{align}
    \left.\frac{\mathrm{d}\tau}{\mathrm{d}z}\right|_{z=0} &= -ifU_g,
    \label{eq:NewBoundaryCondition1}
    \\
    \tau(z=H) &= 0.
    \label{eq:NewBoundaryCondition2}
\end{align}
\end{subequations}
Above, $H$ becomes the height of the boundary layer, and the upper boundary condition was argued from the jump condition in the two-layer model, as seen in \autoref{eq:JumpCondition}.

\subsubsection{Simple viscosity profiles for the SUBC model}
Using the the simplified upper boundary condition model, SUBC model, one can study some simple viscosity profiles to be evaluated numerically. The easiest, except for the constant viscosity, is a linearly varying viscosity with height as 
\begin{subequations}
    \label{eq:LinearProfiles}   
 \begin{align}
    \label{eq:LinearIncreasing}
    \nu_\uparrow(z)&=\frac{\nu_\text{max}}{H}z, \\
    \label{eq:LinearDecreasing}
    \nu_\downarrow(z)&=\frac{\nu_\text{max}}{H}\left(H-z\right), 
\end{align}
\end{subequations}
where $\nu_\text{max}$ is the maximal viscosity. For both of these profiles one can define a dimensionless height coordinate as 
\begin{equation}
    \tilde z = \frac{z}{H} \implies 
    \begin{cases}
        \nu_\uparrow(z)=\nu_\text{max}\tilde z \\
        \nu_\downarrow(z)=\nu_\text{max}\left(1-\tilde z\right). 
    \end{cases}
\end{equation}
One must note that this new coordinate changes the previous $z$-derivative, as a factor of $1/H$ appears when taking the derivative with respect to $\tilde z$. Using these linear viscosity profiles, together with the new dimensionless height coordinate, one can rewrite \autoref{eq:MomentumFlux} for the two profiles as 
\begin{subequations}
\label{eq:15Linear}
\begin{align}
    \label{eq:15LinearIncreasing}
    \frac{\mathrm{d}^2\tau}{\mathrm{d} \tilde z^2}&=i\underbrace{\frac{1}{ \tilde z}}_{\equiv 1/\tilde\nu_\uparrow}\underbrace{\frac{Hf^2}{\nu_\text{max}}}_{=2\varphi^2}\tau,\\
    \label{eq:15LinearDecreasing}
    \frac{\mathrm{d}^2\tau}{\mathrm{d} \tilde z^2}&=i\underbrace{\frac{1}{ 1-\tilde z}}_{\equiv 1/\tilde\nu_\downarrow}\underbrace{\frac{Hf^2}{\nu_\text{max}}}_{=2\varphi^2}\tau.
\end{align}
\end{subequations}
Above the dimensionless height $\varphi$ was again introduced following the logic from before.

Another simple profile to evaluate is a parabolic increasing viscosity as 
\begin{equation}
    \nu_p=\frac{4\nu_\text{max}}{H^2}\left(H-z\right)z.
    \label{eq:ParabolicProfile}
\end{equation} 
Again, using the non-dimensional height coordinate $\tilde z$ and the dimensionless height $\varphi$, one can rewrite \autoref{eq:EkmanEquation} as 
\begin{equation}
    \frac{\mathrm{d}^2\tau}{\mathrm{d} \tilde z^2}=i \underbrace{\frac{1}{ 4\tilde z \left[1-\tilde z\right]}}_{\equiv 1/\tilde\nu_p} \underbrace{\frac{Hf^2}{\nu_\text{max}}}_{=2\varphi^2}(U-U_g).
\end{equation}


\subsection{The Ekman transport}
For a theoretically or numerically derived solution, one can derive the Ekman transport, which may be a more useful quantity. The Ekman transport is defined as the mass flux per meter for the entire Ekman spiral. In the classical solution the Ekman transport is \ang{90} to the right of the surface flow, and can be obtained by integrating the the velocity solution with altitude. One can thus define the Ekman transport as 
\begin{equation}
    T(\varphi) = \int_0^\infty\left[U-U_g\right]d z=\frac{i}{f}\tau(z=0),
    \label{eq:Transport}
\end{equation}
where the definition of the vertical momentum flux (see \autoref{eq:UtoTau}) was used.To obtain the transport angle one can simply compute the argument, $ \arg\left[T(\varphi)\right] $ of the transport.
